(async () => {
  await document.fonts.ready;
  if ([...document.fonts].some(font => font.status === "error")) throw new Error("A document font failed to load");
  await Promise.all([...document.images].map(image => image.decode()));
  // Navigation waits for load, including stylesheets and CSS images. Let the
  // browser parse CSS URLs; reparsing computed styles breaks escaped filenames.
  await new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(resolve)));
  return Math.max(document.documentElement.scrollHeight, document.body?.scrollHeight || 0, 1);
})()
